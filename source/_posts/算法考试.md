int b[100001];//存质数 
long long n;
int main()
{
    int cnt = 0;
    cin >> n;//看你要查的范围是多大啦。
    for (int i = 2; i <= n; i++)
    {
        if (a[i] == 0)    b[++cnt] = i;
        for (int j = 1; j <= cnt; j++)
        {
            if (i * b[j] > n)break;// 如果超出给出的范围，那么就退出循环 
            a[i * b[j]] = 1;//素数的倍数不是素数，进行标记。
            if (i % b[j] == 0)break;//超级关键的只标记一次
        }
    }
