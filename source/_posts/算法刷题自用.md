并查集板子

``` java
void union(int a,int b){
	p[find(a)] = p[find(b)];
}

int find(int x){
	if(p[x]!=x) p[x] = find(p[i]);
	return p[i];
}
boolean query(int a,int b){
	return p[find(a)] == p[find(b)];
}
```
